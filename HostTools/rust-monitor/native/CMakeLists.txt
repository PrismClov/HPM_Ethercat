cmake_minimum_required(VERSION 3.28)
project(ecat_monitor_backends LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(MONITOR_SOEM "Build the SOEM 2.0.0 backend" ON)
option(MONITOR_IGH "Build the IgH backend" OFF)

if(MONITOR_SOEM)
    if(SOEM_ROOT)
        set(SOEM_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
        add_subdirectory("${SOEM_ROOT}" soem)
    else()
        find_package(soem 2.0.0 EXACT CONFIG REQUIRED)
    endif()
    add_executable(ecat-monitor-soem soem.c)
    target_link_libraries(ecat-monitor-soem PRIVATE soem)
endif()

if(MONITOR_IGH)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "IgH requires Linux")
    endif()
    find_path(IGH_INCLUDE_DIR ecrt.h HINTS /opt/etherlab PATH_SUFFIXES include REQUIRED)
    find_library(IGH_LIBRARY ethercat HINTS /opt/etherlab PATH_SUFFIXES lib lib64 REQUIRED)
    add_executable(ecat-monitor-igh igh.c)
    target_include_directories(ecat-monitor-igh PRIVATE "${IGH_INCLUDE_DIR}")
    target_link_libraries(ecat-monitor-igh PRIVATE "${IGH_LIBRARY}")
endif()
