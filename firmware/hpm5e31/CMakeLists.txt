cmake_minimum_required(VERSION 3.13)

set(HPM_PDO_MODE phase CACHE STRING "HPM PDO firmware mode: phase or echo")
set_property(CACHE HPM_PDO_MODE PROPERTY STRINGS phase echo)
if(NOT HPM_PDO_MODE MATCHES "^(phase|echo)$")
    message(FATAL_ERROR "HPM_PDO_MODE must be phase or echo.")
endif()

if(NOT DEFINED HPM_BUILD_TYPE OR HPM_BUILD_TYPE STREQUAL "")
    if(HPM_PDO_MODE STREQUAL "echo")
        set(HPM_BUILD_TYPE ram)
    else()
        set(HPM_BUILD_TYPE flash_xip)
    endif()
endif()

set(BOARD hpm5e00kit CACHE STRING "HPM board")
set(BOARD_SEARCH_PATH "${CMAKE_CURRENT_LIST_DIR}/boards" CACHE PATH "Custom board directory")
set(CONFIG_EEPROM_EMULATION 1)
set(CONFIG_SEGGER_RTT 1 CACHE BOOL "Enable the board debug console over RTT")

set(LUCKYCAT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../ecat-ram/LuckyCAT_SoftWare-main"
    CACHE PATH "Path to the LuckyCAT sample and its locally available SSC source")
get_filename_component(luckycat_root "${LUCKYCAT_ROOT}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(sample_dir "${luckycat_root}/demo/4_ethercat_io")
set(ssc_dir "${sample_dir}/ecat_io/SSC/Src")
set(ssc_application "${sample_dir}/ecat_io/application/digital_io.c")
set(oled_dir "${sample_dir}/OLED")
set(ssc_sources coeappl.c ecatappl.c ecatcoe.c ecatslv.c mailbox.c objdef.c sdoserv.c)
set(required_files "${ssc_application}" "${ssc_dir}/digital_ioObjects.h" "${oled_dir}/oled.h" "${oled_dir}/oled.c")
foreach(source IN LISTS ssc_sources)
    list(APPEND required_files "${ssc_dir}/${source}")
endforeach()
foreach(file IN LISTS required_files)
    if(NOT EXISTS "${file}")
        message(FATAL_ERROR "Missing ${file}. Set LUCKYCAT_ROOT to the local LuckyCAT sample tree. See firmware/hpm5e31/README.md.")
    endif()
endforeach()

find_package(hpm-sdk REQUIRED HINTS $ENV{HPM_SDK_BASE})
project(hpm5e31_ethercat)

configure_file(src/ssc_application.c.in "${CMAKE_CURRENT_BINARY_DIR}/ssc_application.c" @ONLY)
sdk_inc("${ssc_dir}" src port)
foreach(source IN LISTS ssc_sources)
    sdk_src("${ssc_dir}/${source}")
endforeach()
sdk_src("${CMAKE_CURRENT_BINARY_DIR}/ssc_application.c")
sdk_src(src/pdo_application.c port/hpm_ecat_hw.c port/hpm_ecat_phy.c port/hpm_ecat_e2p_emulation.c)
# The PDO hooks are in the SDK library, so its compiler also needs OLED headers.
sdk_inc("${oled_dir}")
sdk_app_src(src/ecat.c "${oled_dir}/oled.c")
if(HPM_PDO_MODE STREQUAL "echo")
    sdk_compile_definitions(-DHPM_PDO_ECHO=1)
endif()
